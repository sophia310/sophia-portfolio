auth.middleware.js
export const verifyAuth = (req, res, next) => {
  const token = req.cookies.token;
  if (!token) return res.status(401).end();
  jwt.verify(token, process.env.JWT_SECRET, (err, user) => {
    req.user = user;
    next();
  });
};
